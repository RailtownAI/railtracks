324        self, schema: DecisionSchema, answers: Mapping[str, DecisionAnswer]
325    ) -> None:
326        """Hold one answer per question.
327
328        Args:
329            schema: The schema the answers are for.
330            answers: The answers keyed by question name.
331
332        Raises:
333            ValueError: If an answer is missing, unexpected, or of the wrong type.
334        """
335        questions = schema.questions
336        unexpected = set(answers) - set(questions)
337        if unexpected:
338            raise ValueError(f"Unexpected answers: {sorted(unexpected)}")
339        for name, question in questions.items():
340            if name not in answers:
341                raise ValueError(f"Missing answer for question {name!r}")
342            if not isinstance(answers[name], question.answer_type):
343                raise ValueError(
344                    f"Answer for question {name!r} must be a {question.answer_type.__name__}, got {type(answers[name]).__name__}"
345                )
346        self._schema = schema
347        self._answers = {name: answers[name] for name in questions}

183    async def _adecisions(
184        self, state: DecisionInput, schema: DecisionSchema
185    ) -> LiteLLMDecisionResponse:
186        """One ``litellm.adecisions`` call, retried per ``retry_approach``."""
187
188        async def adecisions() -> LiteLLMDecisionResponse:
189            return await litellm.adecisions(
190                model=self.litellm_model,
191                input=to_input(state),
192                questions=questions_to_wire(schema),
193                api_key=self._api_key,
194                api_base=self._api_base,
195                timeout=self._timeout,
196            )
197
198        try:
199            if self.retry_approach is None:
200                return await adecisions()
201            return await self.retry_approach.acall_with_retry(adecisions)
202        except Exception as e:
203            # classify once, by the attempt the retries ended on
204            failure = e.exception_list[-1] if isinstance(e, RetryError) else e
205            error = self._provider_error(failure)
206            if error is None:
207                raise
208            if isinstance(e, RetryError):
209                error.notes.append(f"Gave up after {len(e.exception_list)} attempts.")
210            raise error from failure

216                return provider_type(
